  <p class="note" id="time"></p>


  <script>

    // 1) Frontend ส่งคำขอ (GET) ไปที่ API

    fetch('https://open.er-api.com/v6/latest/USD')

      // 2) แปลงคำตอบที่เป็น JSON ให้ใช้งานได้

      .then(res => res.json())

      // 3) หยิบเฉพาะค่า THB มาแสดงบนหน้าเว็บ

      .then(data => {

        document.getElementById('rate').textContent =

          data.rates.THB.toFixed(2) + ' บาท';

        document.getElementById('time').textContent =

          'ข้อมูลอัปเดตเมื่อ: ' + data.time_last_update_utc;

      })

      // 4) ถ้า API ไม่ตอบ ให้บอกผู้ใช้

      .catch(() => {

        document.getElementById('rate').textContent = 'ดึงข้อมูลไม่สำเร็จ';

      });

  </script>
